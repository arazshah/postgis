    منبع و مجوز

    داده‌ها از OpenStreetMap هستند و با مجوز ODbL 1.0 منتشر می‌شوند. در هر نقشه یا گزارشی که از آن‌ها می‌سازید، منبع را ذکر کنید: «© OpenStreetMap contributors». اگر پایگاه داده‌ای مشتق از آن را عمومی کنید، باید با همان مجوز باشد. کامل بودن داده در شهرهای مختلف یکسان نیست.

    نمونه: کوتاه‌ترین مسیر

    نزدیک‌ترین گره شبکه به دو نقطه را پیدا می‌کند و مسیر بین آن‌ها را با طول هر قطعه برمی‌گرداند (در «اجرای SQL» یا QGIS):

    WITH a AS (SELECT id FROM routing.ways_vertices_pgr
           ORDER BY the_geom <-> ST_SetSRID(ST_MakePoint(51.389, 35.700), 4326) LIMIT 1),
     b AS (SELECT id FROM routing.ways_vertices_pgr
           ORDER BY the_geom <-> ST_SetSRID(ST_MakePoint(51.420, 35.760), 4326) LIMIT 1)
SELECT r.seq, w.name, w.length_m, w.the_geom
FROM pgr_dijkstra('SELECT gid AS id, source, target, cost, reverse_cost FROM routing.ways',
                  (SELECT id FROM a), (SELECT id FROM b)) r
JOIN routing.ways w ON w.gid = r.edge;
