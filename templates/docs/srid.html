تبدیل سیستم مختصات

-- محاسبه مساحت به متر مربع در تهران
SELECT id, ST_Area(ST_Transform(geom, 32639)) AS area_m2 FROM parcels;

-- اصلاح SRID اشتباه (فقط برچسب را عوض می‌کند، مختصات را نه)
SELECT UpdateGeometrySRID('parcels', 'geom', 32639);
قاعده ساده: داده را با SRID اصلی‌اش ذخیره کنید، برای محاسبه فاصله و مساحت به UTM ببرید، و برای نمایش وب به 3857 یا 4326.
