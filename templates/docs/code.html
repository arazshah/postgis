Python با psycopg

import psycopg

with psycopg.connect("host=HOST port=5432 dbname=DB user=USER password=PASS sslmode=require") as conn:
    rows = conn.execute("SELECT name, ST_AsGeoJSON(geom) FROM places LIMIT 10").fetchall()
    for name, geojson in rows:
        print(name, geojson)

import geopandas as gpd

Node.js با node-postgres

import pg from "pg";

const pool = new pg.Pool({
  connectionString: "postgresql://USER:PASS@HOST:5432/DB",
  ssl: { rejectUnauthorized: false },
  max: 3,
});

const { rows } = await pool.query(
  "SELECT id, ST_AsGeoJSON(geom)::json AS geometry FROM places WHERE ST_DWithin(geom::geography, ST_MakePoint($1,$2)::geography, 1000)",
  [51.389, 35.689]
);
