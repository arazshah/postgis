-- schools within 500 m of a point
SELECT name,
  ST_Distance(geom::geography, p) AS m
FROM schools,
  ST_MakePoint(45.07, 37.55)::geography p
WHERE ST_DWithin(geom::geography, p, 500)
ORDER BY m;

-- 6 rows · 4 ms
