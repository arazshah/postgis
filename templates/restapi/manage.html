نمونه استفاده

جدول را انتخاب کنید تا نمونه‌ها با نام و کلید اصلی همان جدول عوض شود. آدرس پایه: {{ base }}/rest/{{ service.slug }}/. کلید را با هدر Authorization: Bearer بفرستید؛ خروجی جدول مکانی GeoJSON است.

{% for a in apis %}{% endfor %} 
# فهرست با فیلتر، مرتب‌سازی و محدوده
curl -H "Authorization: Bearer $KEY" \
  "{{ base }}/rest/{{ service.slug }}/{{ sample.path }}/?limit=20&order=-{{ pk_of|get_item:sample.pk }}&bbox=51.2,35.6,51.6,35.8"

# فیلترها: ستون=مقدار، ستون__gt، __gte، __lt، __lte، __ne، __like (با *)، __in (با ویرگول)، __null
curl -H "Authorization: Bearer $KEY" "{{ base }}/rest/{{ service.slug }}/{{ sample.path }}/?name__like=*پارک*"

# افزودن سطر (GeoJSON Feature یا JSON ساده)
curl -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"type":"Feature","geometry":{"type":"Point","coordinates":[51.39,35.70]},"properties":{"name":"نمونه"}}' \
  "{{ base }}/rest/{{ service.slug }}/{{ sample.path }}/"

# خواندن، ویرایش و حذف یک سطر با کلید اصلی
curl -X PATCH -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"name":"نام جدید"}' "{{ base }}/rest/{{ service.slug }}/{{ sample.path }}/12/"
curl -X DELETE -H "Authorization: Bearer $KEY" "{{ base }}/rest/{{ service.slug }}/{{ sample.path }}/12/"
// JavaScript در مرورگر یا اپلیکیشن
const r = await fetch("{{ base }}/rest/{{ service.slug }}/{{ sample.path }}/?limit=100",
                      {headers: {Authorization: "Bearer " + KEY}});
const data = await r.json();

کلید دارای اجازه‌ی نوشتن را فقط در سرور یا اپلیکیشنی بگذارید که در اختیار خودتان است، نه در کد صفحه‌ای که همه می‌بینند؛ برای جمع‌آوری داده از عموم، فرم میدانی بسازید.
