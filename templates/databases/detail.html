# pip install geopandas sqlalchemy psycopg2-binary
import geopandas as gpd
from sqlalchemy import create_engine

engine = create_engine(
    "postgresql+psycopg2://{{ db.role_name }}:{{ pw }}@{{ db.host }}:{{ db.port }}/{{ db.db_name }}?sslmode={{ db.sslmode }}"
)

gdf = gpd.read_file("parcels.shp")
gdf.to_postgis("parcels", engine, if_exists="replace")

print(gpd.read_postgis("SELECT * FROM parcels LIMIT 5", engine, geom_col="geometry"))

با ogr2ogr (همراه QGIS و GDAL نصب می‌شود):

ogr2ogr -f PostgreSQL \
  PG:"host={{ db.host }} port={{ db.port }} dbname={{ db.db_name }} user={{ db.role_name }} password={{ pw }} sslmode={{ db.sslmode }}" \
  parcels.shp -nln parcels -nlt PROMOTE_TO_MULTI -lco GEOMETRY_NAME=geom -t_srs EPSG:4326 -progress
